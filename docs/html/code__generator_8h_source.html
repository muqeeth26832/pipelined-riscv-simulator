47 friend std::ostream &operator<<(std::ostream &os, const ICUnit &unit) {
48 auto emit_if_filled = [&os](const std::array<char, 6> &field,
49 bool &first_operand) {
50 if (field[0] != '\0') {
51 os << (first_operand ? " " : ", ")
52 << field.data();
53 first_operand = false;
54 }
55 };
56
57 // 1. line number + opcode
58 os
59 // << std::hex
60 // << unit.instruction_index * 4
61 // << std::setw(0) << std::dec
62 // << ": "
63 << unit.opcode.data();
64
65 // 2. operands
66 bool first = true;
67 emit_if_filled(unit.rd, first);
68 emit_if_filled(unit.rs1, first);
69 emit_if_filled(unit.rs2, first);
70 emit_if_filled(unit.rs3, first);
71
72 // 3. immediates (33‑byte array so treat separately)
73 if (unit.imm[0] != '\0') {
74 os << (first ? " " : ", ") << unit.imm.data();
75 first = false;
76 }
77
78 // 4. CSR (print only if non‑zero)
79 if (unit.csr != 0) {
80 std::ios_base::fmtflags f(os.flags()); // save stream flags
81 os << " csr=0x" << std::hex << unit.csr; // hex looks nicer
82 os.flags(f); // restore flags
83 }
84
85 // 5. rounding mode (print only if non‑zero)
86 if (unit.rm != 0) {
87 os << " rm=" << static_cast<int>(unit.rm);
88 }
89
90 // 6. label (if any) — put at the end in angle brackets
91 if (!unit.label.empty()) {
92 os << " <" << unit.label << '>';
93 }
94
95 return os;
96 }
